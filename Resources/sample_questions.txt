Sample Question

1. Total sum of debit for each account owner

select a.name, sum(t.debit) from account a inner join transaction t on a.id = t.account_id group by (a.name,t.debit);

2. Total credit and total debit between '2026-11-01' AND '2026-11-06';
SELECT
    SUM(debit) AS total_debit,
    SUM(credit) AS total_credit
FROM transaction
WHERE date BETWEEN '2026-11-01' AND '2026-11-06';

3 list total debit of each account holder
SELECT a.name AS account_name,
    SUM(t.debit) AS total_debit
FROM account a JOIN transaction t
    ON a.id = t.account_id
GROUP BY a.id, a.name
ORDER BY total_debit DESC;

4. Maximum credit added by customer in descending order 
4. min debit added by customer in descending order 
select a.name , min(t.debit) AS mindebit from account a JOIN transaction t on a.id = t.account_id GROUP BY t.debit, a.name  order by mindebit;
